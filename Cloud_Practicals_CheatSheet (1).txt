4)Factorial using while loop 

import java.util.Scanner;

class Factorial {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();

        int fact = 1;
        int i = 1;

        while(i <= n) {
            fact *= i;
            i++;
        }

        System.out.println("Factorial = " + fact);
    }
}

5) Biggest among three (if-else)

import java.util.Scanner;

class BiggestIf {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();

        if(a > b && a > c)
            System.out.println("Biggest = " + a);
        else if(b > c)
            System.out.println("Biggest = " + b);
        else
            System.out.println("Biggest = " + c);
    }
}

6) biggest among three ( ternary)
import java.util.Scanner;

class BiggestTernary {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int a = sc.nextInt();
        int b = sc.nextInt();
        int c = sc.nextInt();

        int max = (a > b) ? (a > c ? a : c) : (b > c ? b : c);
        System.out.println("Biggest = " + max);
    }
}

7)Grade using switch case

import java.util.Scanner;

class GradeSwitch {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int marks = sc.nextInt();
        int grade = marks / 10;

        switch(grade) {
            case 10:
            case 9: System.out.println("A"); break;
            case 8: System.out.println("B"); break;
            case 7: System.out.println("C"); break;
            case 6: System.out.println("D"); break;
            default: System.out.println("Fail");
        }
    }
}

8️⃣ Dynamic initialization – Average of n numbers

import java.util.Scanner;

class AverageN {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int sum = 0;

        for(int i = 0; i < n; i++) {
            sum += sc.nextInt();
        }

        double avg = (double)sum / n;
        System.out.println("Average = " + avg);
    }
}

class MatrixAdd {
    public static void main(String[] args) {
        int a[][] = {{1,2},{3,4}};
        int b[][] = {{5,6},{7,8}};
        int c[][] = new int[2][2];

        for(int i=0;i<2;i++) {
            for(int j=0;j<2;j++) {
                c[i][j] = a[i][j] + b[i][j];
                System.out.print(c[i][j] + " ");
            }
            System.out.println();
        }
    }
}

1️⃣4️⃣ Average using method overloading

class AvgOverload {

    void avg(int a, int b, int c) {
        System.out.println("Average = " + (a+b+c)/3);
    }

    void avg(double a, double b, double c) {
        System.out.println("Average = " + (a+b+c)/3);
    }

    public static void main(String[] args) {
        AvgOverload obj = new AvgOverload();
        obj.avg(3,6,9);
        obj.avg(3.5,6.5,9.5);
    }
}

1️⃣5️⃣ Average using method overriding

class Parent {
    void avg(int a, int b, int c) {
        System.out.println("Average = " + (a+b+c)/3);
    }
}

class Child extends Parent {
    void avg(int a, int b, int c) {
        System.out.println("Average (Overridden) = " + (a+b+c)/3);
    }

    public static void main(String[] args) {
        Child obj = new Child();
        obj.avg(3,6,9);
    }
}

1️⃣7️⃣ String palindrome

import java.util.Scanner;

class Palindrome {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String str = sc.next();
        String rev = "";

        for(int i = str.length()-1; i >= 0; i--) {
            rev += str.charAt(i);
        }

        if(str.equals(rev))
            System.out.println("Palindrome");
        else
            System.out.println("Not Palindrome");
    }
}
